82template<typename T>
83bool is_zero_density(const T& rho) {
84 bool is_zero = false;
85 bool is_positive = false;
86 if constexpr(requires {
87 rho.lower();
88 rho.upper();
89 }) { // Intervals
90 is_zero = rho.lower() == 0.0 && rho.upper() == 0.0;
91 is_positive = rho.lower() > 0.0;
92 } else if constexpr(requires {
93 rho.mean();
94 rho.sd();
95 }) { // Uncertain
96 is_zero = rho.mean() == 0.0 && rho.sd() == 0.0;
97 is_positive = rho.mean() > 0.0;
98 } else {
99 is_zero = rho == 0.0;
100 is_positive = rho > 0.0;
101 }
102 if(!is_zero && !is_positive)
103 throw std::domain_error("SVWN3: density must be exactly zero or "
104 "strictly positive");
105 return is_zero;
106}

123template<typename T>
124T vwn_eps(const T& x) {
125 using std::atan;
126 using std::log;
127 const T X = x * x + b * x + c;
128 const T x_m_x0 = x - x0;
129 const T term1 = log(x * x / X);
130 const T term2 = atan_coef() * atan(Q() / (2.0 * x + b));
131 const T term3 = f2() * log(x_m_x0 * x_m_x0 / X);
132 const T eps = A * (term1 + term2 - term3);
133 return eps;
134}

137template<typename T>
138T vwn_deps_dx(const T& x) {
139 const T X = x * x + b * x + c;
140 const T two_x_b = 2.0 * x + b;
141 const T dterm1 = 2.0 / x - two_x_b / X;
142 const T dterm2 =
143 atan_coef() * (-2.0 * Q()) / (two_x_b * two_x_b + Q() * Q());
144 const T dterm3 = f2() * (2.0 / (x - x0) - two_x_b / X);
145 return A * (dterm1 + dterm2 - dterm3);
146}
